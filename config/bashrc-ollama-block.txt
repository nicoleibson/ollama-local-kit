ofree() {
    OLLAMA_KIT_LOCAL=1 \
    OLLAMA_KIT_PROVIDER=ollama \
    OLLAMA_HOST= OLLAMA_API_KEY= NVIDIA_API_KEY= GEMINI_API_KEY= \
        $HOME/ollama-cli.py "$@"
}
ocloud() {
    if [ ! -s "$HOME/.ollama-cloud-key" ]; then
        echo "Falta ~/.ollama-cloud-key con tu API key de ollama.com" >&2
        return 1
    fi
    OLLAMA_HOST=https://ollama.com \
    OLLAMA_API_KEY="$(cat "$HOME/.ollama-cloud-key")" \
        $HOME/ollama-cli.py "$@"
}
onvidia() {
    if [ ! -s "$HOME/.nvidia-api-key" ]; then
        echo "Falta ~/.nvidia-api-key con tu API key de NVIDIA NIM" >&2
        return 1
    fi
    OLLAMA_HOST= \
    OLLAMA_KIT_PROVIDER=nvidia \
    NVIDIA_API_KEY="$(cat "$HOME/.nvidia-api-key")" \
        $HOME/ollama-cli.py "$@"
}
ogemini() {
    if [ ! -s "$HOME/.gemini-api-key" ]; then
        echo "Falta ~/.gemini-api-key con tu API key de Gemini" >&2
        return 1
    fi
    OLLAMA_HOST= \
    OLLAMA_KIT_PROVIDER=gemini \
    GEMINI_API_KEY="$(cat "$HOME/.gemini-api-key")" \
        $HOME/ollama-cli.py "$@"
}
